async function loadOHLC() { const timeframe = document.getElementById("timeframe").value; const intervalMap = { "1M": "1min", "5M": "5min", "15M": "15min", "1H": "1h" }; const interval = intervalMap[timeframe] || "15min"; const url = "https://api.twelvedata.com/time_series" + "?symbol=XAU/USD" + "&interval=" + encodeURIComponent(interval) + "&outputsize=100" + "&timezone=Africa/Nairobi" + "&order=desc" + "&apikey=" + encodeURIComponent(TWELVE_DATA_KEY) + "&_=" + Date.now(); try { const response = await fetch(url, { cache: "no-store" }); const data = await response.json(); if ( !response.ok || data.status === "error" || !Array.isArray(data.values) ) { throw new Error( data.message || "No valid candle data returned" ); } const candles = data.values .map(item => ({ datetime: item.datetime, open: Number(item.open), high: Number(item.high), low: Number(item.low), close: Number(item.close) })) .filter(item => item.datetime && Number.isFinite(item.open) && Number.isFinite(item.high) && Number.isFinite(item.low) && Number.isFinite(item.close) && item.high >= item.low && item.open >= item.low && item.open <= item.high && item.close >= item.low && item.close <= item.high ); if (!candles.length) { throw new Error("No valid candles returned"); } candles.sort((a, b) => { const dateA = new Date(a.datetime.replace(" ", "T")); const dateB = new Date(b.datetime.replace(" ", "T")); return dateA - dateB; }); const latest = candles[candles.length - 1]; const latestTime = new Date( latest.datetime.replace(" ", "T") ); const ageMinutes = (Date.now() - latestTime.getTime()) / 60000; const allowedAge = { "1M": 8, "5M": 15, "15M": 35, "1H": 130 }[timeframe] || 35; document.getElementById("ohlcStatus").textContent = "LIVE"; document.getElementById("ohlcStatus").className = "status live"; document.getElementById("candleCount").textContent = candles.length; document.getElementById("latestCandle").textContent = latest.datetime + " | O " + latest.open.toFixed(2) + " H " + latest.high.toFixed(2) + " L " + latest.low.toFixed(2) + " C " + latest.close.toFixed(2); document.getElementById("candleAge").textContent = Number.isFinite(ageMinutes) ? ageMinutes.toFixed(1) + " minutes" : "---"; marketData.candles = candles; marketData.lastCandleTime = latestTime; if ( Number.isFinite(ageMinutes) && ageMinutes >= 0 && ageMinutes <= allowedAge ) { document.getElementById("candleFreshness").textContent = "FRESH"; document.getElementById("candleFreshness").className = "status live"; marketData.candleFresh = true; analyzeMarket(); } else { document.getElementById("candleFreshness").textContent = "STALE"; document.getElementById("candleFreshness").className = "status danger"; marketData.candleFresh = false; showWaitingState( "STALE CANDLE DATA", "Latest candle is " + (Number.isFinite(ageMinutes) ? ageMinutes.toFixed(1) : "an invalid") + " minutes old. Waiting for fresh market data." ); } } catch (error) { console.error("OHLC error:", error); document.getElementById("ohlcStatus").textContent = "ERROR"; document.getElementById("ohlcStatus").className = "status danger"; document.getElementById("candleFreshness").textContent = "ERROR"; document.getElementById("candleFreshness").className = "status danger"; marketData.candleFresh = false; showWaitingState( "OHLC FEED ERROR", error.message || "Unable to retrieve current candles." ); } }
